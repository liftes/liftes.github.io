% A colored revision range created by \ntq{<ID>}{...}.
% Examples:
%   p. 6, line 115
%   p. 6, lines 115--117
%   pp. 6--7, lines 115--121
\NewDocumentCommand{\msloc}{m}{%
  \IfRefUndefinedExpandable{M-rev:#1:StartAnchor}
    {\xrefbad{[unresolved manuscript location]}}
    {%
      \ntcrosslink{M-rev:#1:StartAnchor}{\xrefok{%
      \ifnum
        \getpagerefnumber{M-rev:#1:start}=
        \getpagerefnumber{M-rev:#1:end}
        p.~\pageref*{M-rev:#1:start}%
      \else
        pp.~\pageref*{M-rev:#1:start}--%
        \pageref*{M-rev:#1:end}%
      \fi
      ,\space
      \ifnum
        \getrefnumber{M-rev:#1:start}=
        \getrefnumber{M-rev:#1:end}
        line~\ref*{M-rev:#1:start}%
      \else
        lines~\ref*{M-rev:#1:start}--%
        \ref*{M-rev:#1:end}%
      \fi
      }}%
    }%
}

% A location-only anchor created by \ntqpoint{<ID>}.
\NewDocumentCommand{\mspoint}{m}{%
  \IfRefUndefinedExpandable{M-rev:#1:PointAnchor}
    {\xrefbad{[unresolved manuscript location]}}
    {%
      \ntcrosslink{M-rev:#1:PointAnchor}{\xrefok{%
        p.~\pageref*{M-rev:#1:point},
        line~\ref*{M-rev:#1:point}%
      }}%
    }%
}

% Page-only reference to an ordinary manuscript label.
\NewDocumentCommand{\mspage}{m}{%
  \IfRefUndefinedExpandable{M-#1}
    {\xrefbad{[unresolved manuscript reference]}}
    {\xrefok{p.~\ntxpageref{M-#1}}}%
}

% Numbered manuscript figures.  The optional braced second argument is a panel
% suffix, so \msfig{fig:Example}{a} renders as ``Fig. 1a (p. 2)''.
\NewDocumentCommand{\msfig}{m g}{%
  \IfRefUndefinedExpandable{M-#1}
    {\xrefbad{[unresolved manuscript figure]}}
    {\xrefok{Fig.~\ntxref{M-#1}\IfNoValueTF{#2}{}{#2} (p.~\ntxpageref{M-#1})}}%
}

\NewDocumentCommand{\mseq}{m}{%
  \IfRefUndefinedExpandable{M-#1}
    {\xrefbad{[unresolved manuscript equation]}}
    {\xrefok{Eq.~(\ntxref{M-#1}) (p.~\ntxpageref{M-#1})}}%
}

\NewDocumentCommand{\mstable}{m}{%
  \IfRefUndefinedExpandable{M-#1}
    {\xrefbad{[unresolved manuscript table]}}
    {\xrefok{Table~\ntxref{M-#1} (p.~\ntxpageref{M-#1})}}%
}
